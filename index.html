// Webからのリクエスト受取（データ取得 ＆ 申請登録） function doGet(e) { const params = e.parameter; const action = params.action; // 💡 シフト申請の保存処理 if (action === "submitRequest") { return handleApplyShift(params); } // 💡 カレンダー表示用のデータ取得処理 const userId = params.userId || ""; const displayName = params.displayName || ""; const data = getShiftDataForWeb(userId, displayName); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ========================================== // 📱 WEBカレンダー用データ取得（確定シフト＋可能コード＋自分の申請） // ========================================== function getShiftDataForWeb(lineUserId, lineDisplayName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. スタッフマスタからスタッフ名を特定 let currentStaffName = ""; const staffSheet = ss.getSheetByName("スタッフマスタ"); if (staffSheet) { const staffData = staffSheet.getDataRange().getValues(); for (let i = 1; i < staffData.length; i++) { let name = String(staffData[i][1]).trim(); let uId = String(staffData[i][3]).trim(); let nick = String(staffData[i][4]).trim(); if ((lineUserId && uId === lineUserId) || (lineDisplayName && nick === lineDisplayName)) { currentStaffName = name; break; } } } // 2. 申請可能なシフトコード（申請列が 〇 のもの）を取得 const availableShifts = []; const codeSheet = ss.getSheetByName("シフトコード"); if (codeSheet) { const codeData = codeSheet.getDataRange().getValues(); for (let i = 1; i < codeData.length; i++) { let code = String(codeData[i][0]).trim(); let isAllow = String(codeData[i][4]).trim(); // G列（インデックス4） if (code && (isAllow === "〇" || isAllow === "O" || isAllow === "0")) { availableShifts.push(code); } } } // 3. 「申請倉庫」から自分の申請データを取得 const myRequests = []; const reqSheet = ss.getSheetByName("申請倉庫"); if (reqSheet && currentStaffName) { const reqData = reqSheet.getDataRange().getValues(); for (let i = 1; i < reqData.length; i++) { let sName = String(reqData[i][1]).trim(); let rawDate = reqData[i][2]; let shiftCode = String(reqData[i][3]).trim(); if (sName === currentStaffName && rawDate) { let dateStr = formatDateStr(rawDate); if (dateStr) { myRequests.push({ date: dateStr, shift: shiftCode }); } } } } // 4. シフトデータシートから確定シフトの取得（今月以降＆IS_PUBLISHED=TRUE） const resultEvents = []; const dataSheet = ss.getSheetByName("シフトデータ"); if (dataSheet && dataSheet.getLastRow() >= 2) { const allData = dataSheet.getRange(1, 1, dataSheet.getLastRow(), 4).getValues(); const currentYM = Utilities.formatDate(new Date(), "JST", "yyyy-MM"); const publishedMonths = new Set(); for (let i = 0; i < allData.length; i++) { let ym = formatDateStr(allData[i][0]).substring(0, 7); if (allData[i][1] === "STATUS" && allData[i][2] === "IS_PUBLISHED") { let val = allData[i][3]; let isPublished = (val === true || String(val).toUpperCase().trim() === "TRUE"); if (isPublished && ym && ym >= currentYM) { publishedMonths.add(ym); } } } for (let i = 0; i < allData.length; i++) { let ym = formatDateStr(allData[i][0]).substring(0, 7); if (publishedMonths.has(ym) && allData[i][1] !== "STATUS") { let dateStr = formatDateStr(allData[i][1]); let staffName = String(allData[i][2]).trim(); let shiftType = String(allData[i][3]).trim(); if (dateStr && staffName && shiftType) { resultEvents.push({ start: dateStr, date: dateStr, staff: staffName, shift: shiftType }); } } } } return { myStaffName: currentStaffName, availableShifts: availableShifts, myRequests: myRequests, events: resultEvents }; } // 💡 申請データの書き込み処理 function handleApplyShift(params) { const staffName = params.staffName; const targetDate = params.targetDate; const shiftCode = params.shiftCode; if (!staffName || !targetDate || !shiftCode) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "パラメータ不足" })) .setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActiveSpreadsheet(); let reqSheet = ss.getSheetByName("申請倉庫"); if (!reqSheet) { reqSheet = ss.insertSheet("申請倉庫"); reqSheet.appendRow(["送信日時", "スタッフ名", "希望日付", "希望シフトコード"]); } // 同一スタッフ・同一日付の過去申請があれば更新、なければ末尾追加 const data = reqSheet.getDataRange().getValues(); let updated = false; const now = Utilities.formatDate(new Date(), "JST", "yyyy/MM/dd HH:mm:ss"); for (let i = 1; i < data.length; i++) { let sName = String(data[i][1]).trim(); let dStr = formatDateStr(data[i][2]); if (sName === staffName && dStr === targetDate) { reqSheet.getRange(i + 1, 1).setValue(now); reqSheet.getRange(i + 1, 4).setValue(shiftCode); updated = true; break; } } if (!updated) { reqSheet.appendRow([now, staffName, targetDate, shiftCode]); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } // 日付フォーマット共通関数 function formatDateStr(rawDate) { if (!rawDate) return ""; if (rawDate instanceof Date) { return Utilities.formatDate(rawDate, "JST", "yyyy-MM-dd"); } let str = String(rawDate).trim().replace(/\//g, "-"); let parts = str.split("-"); if (parts.length === 3) { let y = parts[0]; let m = parts[1].padStart(2, '0'); let d = parts[2].padStart(2, '0'); return `${y}-${m}-${d}`; } return ""; }
